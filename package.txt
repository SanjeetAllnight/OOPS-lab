```java
import java.util.Scanner;

/* =========================
   ABSTRACT CLASS
   ========================= */

abstract class Result
{
    abstract void calculateResult();
}

/* =========================
   PARENT CLASS
   ========================= */

class Person
{
    String name;
    int age;

    // Constructor
    Person(String name,int age)
    {
        this.name=name;
        this.age=age;
    }

    void displayPerson()
    {
        System.out.println("Name: "+name);
        System.out.println("Age: "+age);
    }
}

/* =========================
   CHILD CLASS
   Inheritance
   ========================= */

class Student extends Person
{
    int marks[];

    private int rollNo;

    static int studentCount=0;

    int total;
    double percentage;

    // Constructor
    Student(
    String name,
    int age,
    int rollNo,
    int marks[]
    )
    {
        super(name,age);

        this.rollNo=rollNo;

        this.marks=marks;

        studentCount++;
    }

    /* =========================
       Getter Setter
       ========================= */

    public void setRollNo(int rollNo)
    {
        this.rollNo=rollNo;
    }

    public int getRollNo()
    {
        return rollNo;
    }

    /* =========================
       Method Overloading
       ========================= */

    void show()
    {
        System.out.println(
        "Student Information"
        );
    }

    void show(String msg)
    {
        System.out.println(msg);
    }

    /* =========================
       Method
       ========================= */

    void calculate()
    {
        total=0;

        for(int i=0;i<marks.length;i++)
        {
            total+=marks[i];
        }

        percentage=
        total/(double)marks.length;
    }

    /* =========================
       Method Overriding
       ========================= */

    @Override

    void displayPerson()
    {
        super.displayPerson();

        System.out.println(
        "Roll No: "+
        getRollNo()
        );

        System.out.println(
        "Percentage: "+
        percentage
        );
    }

    /* =========================
       Static Method
       ========================= */

    static void totalStudents()
    {
        System.out.println(
        "Total Students: "
        +studentCount
        );
    }
}

/* =========================
   Runtime Polymorphism
   + Abstract Class
   ========================= */

class Report extends Result
{
    Student s;

    Report(Student s)
    {
        this.s=s;
    }

    @Override

    void calculateResult()
    {
        s.calculate();

        if(s.percentage>=40)
        {
            System.out.println(
            "PASS"
            );
        }
        else
        {
            System.out.println(
            "FAIL"
            );
        }
    }
}

/* =========================
   MAIN
   ========================= */

class Blueprint
{
    public static void main(String args[])
    {
        Scanner sc=
        new Scanner(System.in);

        System.out.println(
        "Enter Name:"
        );

        String name=
        sc.nextLine();

        /* String Manipulation */

        name=
        name.toUpperCase();

        System.out.println(
        "Name Length: "
        +name.length()
        );

        System.out.println(
        "First Letter: "
        +name.charAt(0)
        );

        /* Arrays */

        int marks[]=new int[3];

        System.out.println(
        "Enter 3 marks:"
        );

        for(int i=0;i<3;i++)
        {
            marks[i]=
            sc.nextInt();
        }

        Student st=
        new Student(
        name,
        18,
        101,
        marks
        );

        /* Overloading */

        st.show();

        st.show(
        "Welcome Student"
        );

        /* Runtime Polymorphism */

        Result r=
        new Report(st);

        r.calculateResult();

        st.displayPerson();

        /* Static Method */

        Student.totalStudents();

        /* Control Structure */

        if(st.percentage>=75)
        {
            System.out.println(
            "Grade A"
            );
        }
        else if(st.percentage>=50)
        {
            System.out.println(
            "Grade B"
            );
        }
        else
        {
            System.out.println(
            "Grade C"
            );
        }
    }
}
```
